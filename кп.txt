Эта мощная культура, зародившаяся в 1970-х в Бронксе, Нью-Йорк, эволюционировала из локального движения в глобальный феномен, проникая во все сферы жизни, от музыки и моды до языка и искусства.</p>

Рэперы, как современные поэты, часто изобретают и популяризируют сленговые слова, которые затем быстро распространяются среди молодежи, формируя уникальный язык поколения.</p>

    // Ваш JavaScript для игры и поиска будет здесь.

    const questions = [
      {
        question: "Что означает слово 'чилить'?",
        choices: ["Быстро бежать", "Отдыхать, расслабляться", "Врать, обманывать", "Громко смеяться"],
        correctAnswer: "Отдыхать, расслабляться"
      },
      {
        question: "Какое значение у слова 'базара нет'?",
        choices: ["Нет смысла", "Не разговаривай", "Согласен, без вопросов", "Тишина"],
        correctAnswer: "Согласен, без вопросов"
      },
      {
        question: "Что такое 'краш' в современном сленге?",
        choices: ["Серьезная авария", "Мощный компьютер", "Человек, который тебе очень нравится", "Вид кредита"],
        correctAnswer: "Человек, который тебе очень нравится"
      },
      {
        question: "Как расшифровывается 'ROFL'?",
        choices: ["Running Only For Laughs", "Rolling On Floor Laughing", "Really Outrageous Funny Legend", "Right Over Left Foot"],
        correctAnswer: "Rolling On Floor Laughing"
      },
      {
        question: "Что такое 'флексить'?",
        choices: ["Жаловаться", "Хвастаться/демонстрировать", "Прятаться", "Петь"],
        correctAnswer: "Хвастаться/демонстрировать"
      }
    ];

    let currentQuestionIndex = 0;
    let score = 0;

    const questionElement = document.querySelector('.quiz .question');
    const choicesElement = document.querySelector('.quiz .choices');
    const nextButton = document.getElementById('next');
    const scoreElement = document.querySelector('.quiz .score');

    function loadQuestion() {
      const currentQuestion = questions[currentQuestionIndex];
      questionElement.textContent = currentQuestion.question;
      choicesElement.innerHTML = ''; // Очищаем предыдущие варианты

      currentQuestion.choices.forEach(choiceText => {
        const choiceDiv = document.createElement('div');
        choiceDiv.classList.add('choice');
        choiceDiv.textContent = choiceText;
        choicesElement.appendChild(choiceDiv);
      });

      nextButton.textContent = 'Далее';
      nextButton.disabled = true;
      scoreElement.textContent = ''; // Очищаем предыдущий результат
    }

    function selectAnswer(event) {
      const selectedChoice = event.target;
      const correctAnswer = questions[currentQuestionIndex].correctAnswer;

      // Снимаем все предыдущие классы с выборов
      document.querySelectorAll('.quiz .choice').forEach(choice => {
        choice.classList.remove('correct', 'wrong');
        choice.style.pointerEvents = 'none'; // Отключаем клики после выбора
      });

      // Применяем классы к выбранному ответу
      if (selectedChoice.textContent === correctAnswer) {
        selectedChoice.classList.add('correct');
        score++;
      } else {
        selectedChoice.classList.add('wrong');
        // Подсвечиваем правильный ответ, если выбран неправильный
        document.querySelectorAll('.quiz .choice').forEach(choice => {
          if (choice.textContent === correctAnswer) {
            choice.classList.add('correct');
          }
        });
      }

      nextButton.disabled = false;
      scoreElement.textContent = `Счет: ${score} / ${questions.length}`;

      // Проверяем, это последний вопрос
      if (currentQuestionIndex === questions.length - 1) {
        nextButton.textContent = 'Перезапустить';
      } else {
        nextButton.textContent = 'Далее';
      }
    }

    function handleNextButton() {
      if (currentQuestionIndex < questions.length - 1) {
        currentQuestionIndex++;
        loadQuestion();
      } else {
        restartQuiz(); // Перезапускаем игру, если это последний вопрос
      }
    }

    function restartQuiz() {
        currentQuestionIndex = 0;
        score = 0;
        nextButton.textContent = 'Далее';
        nextButton.disabled = true;
        scoreElement.textContent = '';
        loadQuestion(); // Загружаем первый вопрос снова
        // Удаляем "правильный" / "неправильный" классы со всех выборов
        document.querySelectorAll('.quiz .choice').forEach(choice => {
            choice.classList.remove('correct', 'wrong');
            choice.style.pointerEvents = 'auto'; // Включаем клики снова
        });
    }


    // Инициализация игры
    loadQuestion();

    // Обработчики событий
    choicesElement.addEventListener('click', (event) => {
      if (event.target.classList.contains('choice')) {
        selectAnswer(event);
      }
    });

    nextButton.addEventListener('click', handleNextButton);

  </script>
